class Solution {
    ArrayList<Integer> findTwoElement(int arr[]) {
        ArrayList<Integer> ans = new ArrayList<>();
        
        // code here
        int n = arr.length;
        int xr =0;
        
        for(int i=0;i<n;i++){
            xr=xr^arr[i];
            xr=xr^(i+1);
        }
        
        int number =0;
        
        while(true){
            if((xr & (1<<number)) !=0){
                break;
            }else{
                number++;
            }
            
        }
        int one =0;
        int zero =0;
        
        for(int i=0;i<n;i++){
            if((arr[i] & (1<<number))!=0){
                one=one^arr[i];
            }else{
                zero=zero^arr[i];
            }
            
        }
        
        for(int i=1;i<=n;i++){
            if((i & (1<<number))!=0){
                one=one^i;
            }else{
                zero=zero^i;
            }
        }
        
        
        int count =0;
        
        for(int i=0;i<n;i++){
            if(arr[i]==one){
                count++;
            }else{
                continue;
            }
        }
        
        if(count==2){
            ans.add(one);
            ans.add(zero);
        }else{
            ans.add(zero);
            ans.add(one);
        }
        
        return ans;
        
    }
}
